function(add_planetsim_test test_name source_file)
    add_executable(${test_name} ${source_file})
    target_link_libraries(${test_name} PRIVATE PlanetSim::PlanetSim)
    planetsim_set_project_warnings(${test_name})
    planetsim_enable_sanitizers(${test_name})
    add_test(NAME ${test_name} COMMAND ${test_name})
endfunction()
